Алгоритм определения победителя

4raws/pole.cpp — checkWinner()
void pole::checkWinner() {
    // 4 направления: право, вниз, диагональ /, диагональ \
    const int di[] = { 1,  0,  1,  1 };
    const int dj[] = { 0,  1, -1,  1 };

    for (int i = 0; i < WIDTH; i++) {
        for (int j = 0; j < HEIGHT; j++) {
            if (cells[i][j] == Cell::Empty) continue;
            Cell color = cells[i][j];

            for (int d = 0; d < 4; d++) {
                int count = 1;
                for (int k = 1; k < 4; k++) {
                    int ni = i + di[d]*k;
                    int nj = j + dj[d]*k;
                    if (ni < 0 || ni >= WIDTH || nj < 0 || nj >= HEIGHT)
                        break;
                    if (cells[ni][nj] == color) count++;
                    else break;
                }
                if (count == 4) { winner = color; return; }
            }
        }
    }
}

Иерархия фигур

chess/figures.h — полиморфные фигуры
enum class Color     { White, Black };
enum class PieceType { King, Queen, Rook, Bishop, Knight, Pawn };

struct Position { int row, col; };

class Figure {
protected:
    Color     color;
    PieceType type;
    bool      hasMoved = false;

public:
    virtual ~Figure() = default;

    // Полиморфный метод: каждая фигура знает свои ходы
    virtual std::vector<Position> getPseudoMoves(
        int r, int c, const Table& t) const = 0;

    // Для глубокого копирования при симуляции ходов ИИ
    virtual std::unique_ptr<Figure> clone() const = 0;

    virtual char getSymbol() const = 0;
    Color     getColor()    const { return color; }
    PieceType getType()     const { return type; }
    bool      getHasMoved() const { return hasMoved; }
};

// Реализация конкретной фигуры
class Knight : public Figure {
public:
    Knight(Color c) { color = c; type = PieceType::Knight; }
    char getSymbol() const override { return color==Color::White ? 'N' : 'n'; }

    std::unique_ptr<Figure> clone() const override {
        return std::make_unique<Knight>(*this);  // копирование
    }

    std::vector<Position> getPseudoMoves(int r, int c, const Table& t) const override {
        std::vector<Position> moves;
        const int dr[] = {-2,-2,-1,-1, 1, 1, 2, 2};
        const int dc[] = {-1, 1,-2, 2,-2, 2,-1, 1};
        for (int i = 0; i < 8; i++) {
            int nr = r + dr[i], nc = c + dc[i];
            if (nr>=0 && nr<8 && nc>=0 && nc<8)
                if (!t.hasFriendly(nr, nc, color))
                    moves.push_back({nr, nc});
        }
        return moves;
    }
};

ИИ: алгоритм Minimax

ИИ использует алгоритм minimax с альфа-бета отсечением — ищет лучший ход на несколько шагов вперёд:

chess/ai.h — Minimax
class AI {
private:
    int depth;  // глубина поиска
    Color myColor;

    // Оценка позиции: положительная = хорошо для белых
    int evaluate(const Table& t) const;

    // Рекурсивный minimax с альфа-бета отсечением
    int minimax(Table t, int depth, int alpha, int beta, bool maximizing) const;

public:
    AI(Color c, int depth = 3) : myColor(c), depth(depth) {}

    // Возвращает лучший ход
    std::tuple<int,int,int,int> getBestMove(const Table& t) const;
};

int AI::minimax(Table t, int d, int alpha, int beta, bool maximizing) const {
    if (d == 0 || t.isCheckmate(Color::White) || t.isCheckmate(Color::Black))
        return evaluate(t);

    if (maximizing) {
        int maxVal = INT_MIN;
        // Перебираем все ходы белых
        for (auto& [fr,fc,tr,tc] : t.getAllMoves(Color::White)) {
            Table copy = t;
            copy.makeMove(fr, fc, tr, tc);
            maxVal = std::max(maxVal, minimax(copy, d-1, alpha, beta, false));
            alpha = std::max(alpha, maxVal);
            if (beta <= alpha) break;  // альфа-бета отсечение!
        }
        return maxVal;
    } else {
        // Аналогично для минимизирующего игрока...
    }
}

Тест для самопроверки

5 вопросов
Вопрос 1

Зачем в классе Figure есть чисто виртуальный метод clone()?

При копировании Table (для ИИ-симуляции) нужно скопировать все Figure. Но Figure — абстрактный базовый класс, а у нас есть King, Queen и т.д. Виртуальный clone() в каждом подклассе создаёт правильный тип: return make_unique<King>(*this).
Вопрос 2

Почему в алгоритме проверки "4 в ряд" используется только 4 направления, а не 8?

Если проверять "право" и "лево" — одну и ту же линию найдём дважды. Достаточно проверить каждую линию с одного конца: право (→), вниз (↓), диагональ ↗, диагональ ↘. Этого хватает для нахождения всех выигрышных линий.
Вопрос 3

Зачем ИИ делает глубокую копию Table перед симуляцией хода?

Minimax перебирает миллионы вариантов, "делая" ходы и оценивая позицию. Если менять реальную доску, возвращаться назад (undo) было бы сложнее. Копирование доски проще и безопаснее.
Вопрос 4

В чём смысл "альфа-бета отсечения" в minimax?

Альфа-бета: если мы уже нашли ход с оценкой X, и анализируем ветвь, где соперник может получить оценку > X — эту ветвь незачем дальше исследовать. В худшем случае работает как чистый minimax, в лучшем — сокращает дерево поиска на порядки.
Вопрос 5

Метод getValidMoves() отличается от getPseudoMoves(). В чём разница?

getPseudoMoves (в каждом подклассе Figure) даёт все ходы по геометрии фигуры — без учёта шаха. getValidMoves (в Table) проверяет каждый ход: делает копию, выполняет ход, проверяет не оказался ли король под шахом. Только "проходящие" ходы возвращаются как valid.
