Дата и время

Структура класса DateTime

class_DateTime/DateTime.h
class DateTime {
private:
    int year, month, day;
    int hour, minute, second;

    // Вспомогательные статические методы
    static bool isLeapYear(int y);
    static int  daysInMonth(int m, int y);
    static bool isValidDate(int y, int m, int d);

public:
    DateTime(int year, int month, int day,
             int hour=0, int min=0, int sec=0);

    // Геттеры
    int getYear()   const { return year; }
    int getMonth()  const { return month; }
    int getDay()    const { return day; }

    // Вычисления
    int      getDayOfWeek() const;    // формула Зеллера (0=вс, 1=пн...)
    DateTime addDays(int d) const;   // прибавить дни

    // Статические фабричные методы
    static DateTime getEaster(int year);  // алгоритм Гаусса
    static DateTime now();               // текущее время

    // Операторы сравнения
    bool operator<(const DateTime& o) const;
    bool operator==(const DateTime& o) const;

    // Вывод
    friend std::ostream& operator<<(std::ostream&, const DateTime&);
};

Валидация и константы

class_DateTime/DateTime.cpp — валидация
bool DateTime::isLeapYear(int y) {
    // Год високосный если делится на 4, но не на 100,
    // кроме случая деления на 400
    return (y % 4 == 0 && y % 100 != 0) || (y % 400 == 0);
}

int DateTime::daysInMonth(int m, int y) {
    const int days[] = {31,28,31,30,31,30,31,31,30,31,30,31};
    if (m == 2 && isLeapYear(y)) return 29;
    return days[m - 1];
}

DateTime::DateTime(int y, int m, int d, int h, int min, int sec) {
    if (!isValidDate(y, m, d))
        throw std::invalid_argument("Invalid date");
    if (h < 0 || h >= 24 || min < 0 || min >= 60 || sec < 0 || sec >= 60)
        throw std::invalid_argument("Invalid time");
    year=y; month=m; day=d; hour=h; minute=min; second=sec;
}

Формула Зеллера — день недели

Формула Зеллера позволяет вычислить день недели для любой даты Григорианского календаря:

class_DateTime/DateTime.cpp — формула Зеллера
int DateTime::getDayOfWeek() const {
    int m = month, y = year;

    // В формуле январь и февраль — месяцы 13 и 14 предыдущего года
    if (m <= 2) { m += 12; y--; }

    int k = y % 100;      // год в веке
    int j = y / 100;      // номер века

    // Формула Зеллера для Григорианского календаря
    int h = (day + (13*(m+1)/5) + k + (k/4) + (j/4) - 2*j) % 7;

    // h: 0=сб, 1=вс, 2=пн, ... → переводим в 0=вс, 1=пн...
    return ((h + 6) % 7);  // 0=вс, 1=пн, 2=вт... 6=сб
}

Алгоритм Гаусса — дата Пасхи

Пасха вычисляется по сложной формуле. Алгоритм Гаусса — один из вариантов:

class_DateTime/DateTime.cpp — формула Гаусса
DateTime DateTime::getEaster(int year) {
    int a = year % 19;
    int b = year % 4;
    int c = year % 7;

    int k = year / 100;
    int p = (13 + 8*k) / 25;
    int q = k / 4;
    int M = (15 - p + k - q) % 30;
    int N = (4 + k - q) % 7;

    int d = (19*a + M) % 30;
    int e = (2*b + 4*c + 6*d + N) % 7;

    // Результат: 22 марта + d + e дней
    int easterDay   = 22 + d + e;
    int easterMonth = 3;

    if (easterDay > 31) {
        easterDay -= 31;
        easterMonth = 4;
    }
    // Два особых случая...
    if (d == 29 && e == 6) { easterDay = 19; easterMonth = 4; }
    if (d == 28 && e == 6 && a > 10) { easterDay = 18; easterMonth = 4; }

    return DateTime(year, easterMonth, easterDay);
}

Прибавление дней с переходом через месяцы и годы

class_DateTime/DateTime.cpp — addDays
DateTime DateTime::addDays(int days) const {
    int d = day + days, m = month, y = year;

    // Пока дней больше чем в текущем месяце — переходим к следующему
    while (d > daysInMonth(m, y)) {
        d -= daysInMonth(m, y);
        m++;
        if (m > 12) { m = 1; y++; }
    }
    // Аналогично для отрицательных дней (вычитание)
    while (d <= 0) {
        m--;
        if (m < 1) { m = 12; y--; }
        d += daysInMonth(m, y);
    }
    return DateTime(y, m, d, hour, minute, second);
}

Тест для самопроверки

5 вопросов
Вопрос 1

Какой год является високосным?

Правило: делится на 4 → вероятно високосный; делится на 100 → не високосный; делится на 400 → високосный. 1900 делится на 100 но не на 400 → не високосный. 2000 делится на 400 → високосный. Метод isLeapYear() реализует именно это.
Вопрос 2

Зачем в формуле Зеллера январь и февраль переводятся как "13-й и 14-й месяц предыдущего года"?

Формула Зеллера считает год начинающимся с марта (3-й месяц). Тогда в конце "года" (13=январь, 14=февраль) проще учесть поправку на 29 февраля — она всегда в конце цикла.
Вопрос 3

Метод getEaster() является статическим. Почему?

Статический фабричный метод: DateTime easter = DateTime::getEaster(2025);. Не нужно сначала создавать объект, чтобы вычислить другой. Это устойчивый паттерн для создания объектов со сложной логикой инициализации.
Вопрос 4

Почему addDays() использует цикл, а не просто прибавляет к полю day?

При addDays(365) к 1 декабря нужно корректно пройти через декабрь (31 день), январь (31), февраль (28 или 29)... Цикл правильно учитывает переходы, включая високосные годы. Простое прибавление к числу месяца некорректно.
Вопрос 5

Какой паттерн ООП реализует метод DateTime::now()?

Factory Method (фабричный метод): вместо new DateTime(...) используем DateTime::now(). Это инкапсулирует логику создания (получение системного времени, конвертацию) и даёт понятный, выразительный API.
